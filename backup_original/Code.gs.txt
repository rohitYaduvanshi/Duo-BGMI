const SHEET_NAME = "Teams";

// ── GET: Fetch registered team names ──────────────────────────────────
function doGet(e) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const data = sheet.getDataRange().getValues();
  const teams = [];
  for (let i = 1; i < data.length; i++) {
    const name = data[i][0];
    if (name) teams.push(String(name));
  }
  return ContentService
    .createTextOutput(JSON.stringify({ teams: teams }))
    .setMimeType(ContentService.MimeType.JSON);
}

// ── POST: Register team + upload screenshot to Drive ──────────────────
function doPost(e) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
    const body = JSON.parse(e.postData.contents);

    // ── Screenshot Upload ────────────────────────────────────────────
    let screenshotUrl = "";

    if (body.screenshotBase64 && body.screenshotBase64.length > 0) {
      try {
        // Strip the data-URL prefix if present: "data:image/jpeg;base64,..."
        const raw = body.screenshotBase64;
        const base64Only = raw.includes(",") ? raw.split(",")[1] : raw;
        const mimeType   = body.screenshotMime || "image/jpeg";
        const ext        = mimeType.split("/")[1] || "jpg";
        const fileName   = "payment_" + (body.teamName || "team").replace(/\s+/g, "_") + "_" + Date.now() + "." + ext;

        const decoded  = Utilities.base64Decode(base64Only);
        const blob     = Utilities.newBlob(decoded, mimeType, fileName);

        // Save to Drive root (you can change to a specific folder if needed)
        const file = DriveApp.createFile(blob);
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        screenshotUrl = file.getUrl();
      } catch (driveErr) {
        // If Drive upload fails, continue without screenshot
        screenshotUrl = "Upload failed: " + String(driveErr).slice(0, 80);
      }
    }

    // ── Append Row to Sheet ──────────────────────────────────────────
    sheet.appendRow([
      body.teamName  || "",
      body.p1Name    || "",
      body.p1Uid     || "",
      body.p2Name    || "",
      body.p2Uid     || "",
      body.whatsapp  || "",
      body.utr       || "",
      body.tournament || "",
      screenshotUrl,
      body.timestamp  || new Date().toISOString(),
    ]);

    return ContentService
      .createTextOutput(JSON.stringify({ status: "ok", screenshot: screenshotUrl }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: String(err) }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
